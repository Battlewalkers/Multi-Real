Pawns for i in range(BOARD_SIZE): self.board[1][i] = Piece(PieceType.PAWN, Color.BLACK, 1, i) self.board[6][i] = Piece(PieceType.PAWN, Color.WHITE, 6, i) # Rooks self.board[0][0] = Piece(PieceType.ROOK, Color.BLACK, 0, 0) self.board[0][7] = Piece(PieceType.ROOK, Color.BLACK, 0, 7) self.board[7][0] = Piece(PieceType.ROOK, Color.WHITE, 7, 0) self.board[7][7] = Piece(PieceType.ROOK, Color.WHITE, 7, 7) # Knights self.board[0][1] = Piece(PieceType.KNIGHT, Color.BLACK, 0, 1) self.board[0][6] = Piece(PieceType.KNIGHT, Color.BLACK, 0, 6) self.board[7][1] = Piece(PieceType.KNIGHT, Color.WHITE, 7, 1) self.board[7][6] = Piece(PieceType.KNIGHT, Color.WHITE, 7, 6) # Bishops self.board[0][2] = Piece(PieceType.BISHOP, Color.BLACK, 0, 2) self.board[0][5] = Piece(PieceType.BISHOP, Color.BLACK, 0, 5) self.board[7][2] = Piece(PieceType.BISHOP, Color.WHITE, 7, 2) self.board[7][5] = Piece(PieceType.BISHOP, Color.WHITE, 7, 5) # Queens self.board[0][3] = Piece(PieceType.QUEEN, Color.BLACK, 0, 3) self.board[7][3] = Piece(PieceType.QUEEN, Color.WHITE, 7, 3) # Kings self.board[0][4] = Piece(PieceType.KING, Color.BLACK, 0, 4) self.board[7][4] = Piece(PieceType.KING, Color.WHITE, 7, 4) def get_piece_at(self, x: int, y: int) -> Optional[Piece]: return self.board[x][y] def is_valid_move(self, piece: Piece, new_x: int, new_y: int) -> bool: if not (0 <= new_x < BOARD_SIZE and 0 <= new_y < BOARD_SIZE): return False target_piece = self.board[new_x][new_y] if target_piece and target_piece.color == piece.color: return False # Pawn movement if piece.piece_type == PieceType.PAWN: direction = 1 if piece.color == Color.BLACK else -1 start_row = 1 if piece.color == Color.BLACK else 6 # One step forward if new_y == piece.y and new_x == piece.x + direction and not target_piece: return True # Two steps forward from starting position if new_y == piece.y and new_x == piece.x + 2 * direction and piece.x == start_row and not target_piece: return True # Capture diagonally if abs(new_y - piece.y) == 1 and new_x == piece.x + direction and target_piece and target_piece.color != piece.color: return True # Rook movement (horizontal/vertical) elif piece.piece_type == PieceType.ROOK: if piece.x != new_x and piece.y != new_y: return False step_x = 0 if piece.x == new_x else (1 if new_x > piece.x else -1) step_y = 0 if piece.y == new_y else (1 if new_y > piece.y else -1) current_x, current_y = piece.x + step_x, piece.y + step_y while (current_x, current_y) != (new_x, new_y): if self.board[current_x][current_y] is not None: return False current_x += step_x current_y += step_y return True # Bishop movement (diagonal) elif piece.piece_type == PieceType.BISHOP: if abs(piece.x - new_x) != abs(piece.y - new_y): return False step_x = 1 if new_x > piece.x else -1 step_y = 1 if new_y > piece.y else -1 current_x, current_y = piece.x + step_x, piece.y + step_y while (current_x, current_y) != (new_x, new_y): if self.board[current_x][current_y] is not None: return False current_x += step_x current_y += step_y return True # Queen movement (any direction) elif piece.piece_type == PieceType.QUEEN: if piece.x != new_x and piece.y != new_y and abs(piece.x - new_x) != abs(piece.y - new_y): return False step_x = 0 if piece.x == new_x else (1 if new_x > piece.x else -1) step_y = 0 if piece.y == new_y else (1 if new_y > piece.y else -1) current_x, current_y = piece.x + step_x, piece.y + step_y while (current_x, current_y) != (new_x, new_y): if self.board[current_x][current_y] is not None: return False current_x += step_x current_y += step_y return True # Knight movement (L-shape) elif piece.piece_type == PieceType.KNIGHT: dx = abs(piece.x - new_x) dy = abs(piece.y - new_y) return (dx == 2 and dy == 1) or (dx == 1 and dy == 2) # King movement (one step any direction) elif piece.piece_type == PieceType.KING: dx = abs(piece.x - new_x) dy = abs(piece.y - new_y) return dx <= 1 and dy <= 1 return False def select_piece(self, x: int, y: int): piece = self.get_piece_at(x, y) if piece and piece.color == self.current_turn: self.selected_piece = piece self.calculate_valid_moves() def calculate_valid_moves(self): self.valid_moves = [] if self.selected_piece: for i in range(BOARD_SIZE): for j in range(BOARD_SIZE): if self.is_valid_move(self.selected_piece, i, j): self.valid_moves.append((i, j)) def move_piece(self, new_x: int, new_y: int): if self.selected_piece and (new_x, new_y) in self.valid_moves: # Check for pawn promotion if self.selected_piece.piece_type == PieceType.PAWN: if (self.selected_piece.color == Color.WHITE and new_x == 0) or \ (self.selected_piece.color == Color.BLACK and new_x == 7): self.selected_piece.piece_type = PieceType.QUEEN # Move the piece self.board[new_x][new_y] = self.selected_piece self.board[self.selected_piece.x][self.selected_piece.y] = None self.selected_piece.x = new_x self.selected_piece.y = new_y self.selected_piece.has_moved = True # Check for checkmate/check if self.is_checkmate(): self.game_over = True self.winner = self.current_turn # Switch turn self.current_turn = Color.BLACK if self.current_turn == Color.WHITE else Color.WHITE self.selected_piece = None self.valid_moves = [] def is_checkmate(self) -> bool: # Check if current player's king is in check king = None for row in self.board: for piece in row: if piece and piece.color == self.current_turn and piece.piece_type == PieceType.KING: king = piece break if not king: return True # Check if king is in check for row in self.board: for piece in row: if piece and piece.color != self.current_turn: if self.is_valid_move(piece, king.x, king.y): return True return False def get_all_valid_moves(self, color: Color) -> List[Tuple[int, int, int, int]]: moves = [] for i in range(BOARD_SIZE): for j in range(BOARD_SIZE): piece = self.board[i][j] if piece and piece.color == color: for x in range(BOARD_SIZE): for y in range(BOARD_SIZE): if self.is_valid_move(piece, x, y): moves.append((i, j, x, y)) return moves class ChessBot: def __init__(self, difficulty: int): self.difficulty = difficulty def get_move(self, game: ChessGame) -> Tuple[int, int, int, int]: valid_moves = game.get_all_valid_moves(game.current_turn) if not valid_moves: return None # Simple AI based on difficulty if self.difficulty <= 3: # Random moves return random.choice(valid_moves) elif self.difficulty <= 6: # Prefer captures captures = [move for move in valid_moves if game.board[move[2]][move[3]] is not None] if captures: return random.choice(captures) return random.choice(valid_moves) else: # Try to make best moves best_move = None best_value = -float('inf') for move in valid_moves: # Evaluate move piece = game.board[move[0]][move[1]] target_piece = game.board[move[2]][move[3]] value = 0 if target_piece: # Capture value value += self.get_piece_value(target_piece) # Positional value value += self.get_positional_value(piece, move[2], move[3]) if value > best_value: best_value = value best_move = move if best_move: return best_move return random.choice(valid_moves) def get_piece_value(self, piece: Piece) -> int: values = { PieceType.PAWN: 1, PieceType.KNIGHT: 3, PieceType.BISHOP: 3, PieceType.ROOK: 5, PieceType.QUEEN: 9, PieceType.KING: 100 } return values[piece.piece_type] def get_positional_value(self, piece: Piece, x: int, y: int) -> int: # Simple positional evaluation if piece.piece_type == PieceType.PAWN: return 0.1 * (3 if piece.color == Color.WHITE else -3) elif piece.piece_type == PieceType.KNIGHT: return 0.2 * (x + y - 4) elif piece.piece_type == PieceType.BISHOP: return 0.2 * (7 - abs(x - y)) return 0 class OnlineMode: def __init__(self): self.waiting_players = deque() self.active_games = {} self.lock = threading.Lock() def add_player(self, player_id: str): with self.lock: self.waiting_players.append(player_id) def match_players(self) -> Optional[Tuple[str, str]]: with self.lock: if len(self.waiting_players) >= 2: player1 = self.waiting_players.popleft() player2 = self.waiting_players.popleft() return (player1, player2) return None def create_bot_match(self) -> Tuple[str, str]: bot_name = f"Bot_{random.randint(1000, 9999)}" return ("human", bot_name) class ChessApp: def __init__(self): self.screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("Chess Game - Yellow & Green Theme") self.clock = pygame.time.Clock() self.running = True self.game_mode = "menu" self.chess_game = ChessGame() self.bot_difficulty = 5 self.online_mode = OnlineMode() self.bot = ChessBot(self.bot_difficulty) self.selected_square = None self.hovered_square = None #
