Create a new game game_id = f"game_{len(active_games)+1}" active_games[game_id] = { 'player1': opponent_socket, 'player2': client_socket, 'player1_color': 'white', 'player2_color': 'black', 'current_turn': 'white' } # Notify both players response1 = { 'status': 'connected', 'color': 'white', 'opponent': str(opponent_address) } opponent_socket.sendall(json.dumps(response1).encode('utf-8')) response2 = { 'status': 'connected', 'color': 'black', 'opponent': str(address) } client_socket.sendall(json.dumps(response2).encode('utf-8')) else: #
