self.get_king_moves(board) def get_pawn_moves(self, board: List[List[Optional['Piece']]]): moves = [] direction = 1 if self.color == 'white' else -1 # Forward move if 0 <= self.y + direction < 8 and board[self.x][self.y + direction] is None: moves.append((self.x, self.y + direction)) # Double move from starting position if (self.color == 'white' and self.y == 1) or (self.color == 'black' and self.y == 6): if board[self.x][self.y + 2*direction] is None: moves.append((self.x, self.y + 2*direction)) # Captures for dx in [-1, 1]: nx, ny = self.x + dx, self.y + direction if 0 <= nx < 8 and 0 <= ny < 8: target = board[nx][ny] if target and target.color != self.color: moves.append((nx, ny)) return moves def get_rook_moves(self, board: List[List[Optional['Piece']]]): moves = [] directions = [(1,0), (-1,0), (0,1), (0,-1)] for dx, dy in directions: nx, ny = self.x + dx, self.y + dy while 0 <= nx < 8 and 0 <= ny < 8: target = board[nx][ny] if target is None: moves.append((nx, ny)) elif target.color != self.color: moves.append((nx, ny)) break else: break nx += dx ny += dy return moves def get_knight_moves(self, board: List[List[Optional['Piece']]]): moves = [] offsets = [(2,1), target.color != self.color: moves.append((nx, ny)) return moves def get_bishop_moves(self, board: List[List[Optional['Piece']]]): moves = [] directions = [(1,1), (1,-1), (-1,1), (-1,-1)] for dx, dy in directions: nx, ny = self.x + dx, self.y + dy while 0 <= nx < 8 and 0 <= ny < 8: target = board[nx][ny] if target is None: moves.append((nx, ny)) elif target.color != self.color: moves.append((nx, ny)) break else: break nx += dx ny += dy return moves def get_queen_moves(self, board: List[List[Optional['Piece']]]): return self.get_rook_moves(board) + self.get_bishop_moves(board) def
